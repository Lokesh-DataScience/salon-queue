// frontend/src/api/client.ts
// The live site is served by the same server as the API, so production always uses its own address.
const API_URL = import.meta.env.DEV ? (import.meta.env.VITE_API_URL ?? 'http://localhost:4000') : '';

export const SESSION_EXPIRED_EVENT = 'barber:session-expired';

export class ApiError extends Error {
  status: number;
  fieldErrors?: Record<string, string[] | undefined>;
  constructor(message: string, status: number, fieldErrors?: Record<string, string[] | undefined>) {
    super(message);
    this.status = status;
    this.fieldErrors = fieldErrors;
  }
}

type RefreshOutcome = 'ok' | 'denied' | 'error';
let refreshInFlight: Promise<RefreshOutcome> | null = null;

// Single-flight: ten requests failing at once trigger one refresh, not ten.
function refreshSession(): Promise<RefreshOutcome> {
  refreshInFlight ??= fetch(`${API_URL}/api/auth/barber/refresh`, {
    method: 'POST',
    credentials: 'include',
  })
    .then((res): RefreshOutcome => (res.ok ? 'ok' : res.status === 401 ? 'denied' : 'error'))
    .catch((): RefreshOutcome => 'error')
    .finally(() => {
      refreshInFlight = null;
    });
  return refreshInFlight;
}

export async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const { headers: extraHeaders, ...rest } = init ?? {};
  const send = () =>
    fetch(`${API_URL}${path}`, {
      credentials: 'include',
      ...rest,
      headers: { 'Content-Type': 'application/json', ...(extraHeaders as Record<string, string> | undefined) },
    });

  let res = await send();

  // Barber-only routes: if the access token expired, refresh silently and retry once.
  // (A 401 means the handler never ran, so retrying a POST cannot double-apply it.)
  if (res.status === 401 && path.startsWith('/api/barber/')) {
    const outcome = await refreshSession();
    // 'denied' can simply mean another tab rotated the token first. Its new cookies are
    // already in the shared cookie jar, so retry once for 'ok' and 'denied' alike.
    if (outcome !== 'error') {
      res = await send();
      if (res.status === 401) window.dispatchEvent(new Event(SESSION_EXPIRED_EVENT));
    }
  }

  if (!res.ok) {
    const body = await res.json().catch(() => ({ message: 'Something went wrong.' }));
    throw new ApiError(body.message ?? `Request failed with status ${res.status}`, res.status, body.errors);
  }

  return res.json() as Promise<T>;
}

export function joinSalonQueue(salonId: string, clientKey?: string, serviceIds?: string[]) {
  const body: Record<string, unknown> = {};
  if (clientKey) body.clientKey = clientKey;
  if (serviceIds?.length) body.serviceIds = serviceIds;
  return apiFetch<{ ticket: any }>(`/api/queues/${salonId}/join`, {
    method: 'POST',
    body: JSON.stringify(body),
  }).then((r) => r.ticket);
}

export function fetchTicketStatus(ticketId: string, accessToken: string) {
  return apiFetch<{ ticket: any }>(`/api/queue-tickets/${ticketId}`, {
    headers: { 'X-Ticket-Token': accessToken },
  }).then((r) => r.ticket);
}

export function leaveTicket(ticketId: string, accessToken: string) {
  // Token goes in the body, not the URL, so it stays out of logs and history.
  return apiFetch<{ ticket: { ticketId: string; status: string } }>(
    `/api/queue-tickets/${ticketId}/leave`,
    { method: 'POST', body: JSON.stringify({ accessToken }) },
  ).then((r) => r.ticket);
}

// frontend/src/hooks/useActiveTickets.ts
import { useEffect, useMemo } from 'react';
import { useQueries } from '@tanstack/react-query';
import { fetchTicketStatus } from '../api/tickets';
import { ApiError } from '../api/client';
import { getTickets, removeTicket, type StoredTicket } from '../lib/activeTickets';

const FINISHED = ['COMPLETED', 'SKIPPED', 'CANCELLED'];

export interface ActiveTicket {
  stored: StoredTicket;
  ticket: any | null; // null when the server couldn't be reached to verify
}

// Returns only tickets the server confirms are still active, and prunes dead ones from storage.
export function useActiveTickets(salonId?: string) {
  const stored = useMemo(() => getTickets(salonId), [salonId]);

  const results = useQueries({
    queries: stored.map((t) => ({
      queryKey: ['ticket', t.ticketId], // same key as QueueTracking, so the cache is shared
      queryFn: () => fetchTicketStatus(t.ticketId, t.accessToken),
      retry: false,
      staleTime: 15_000,
    })),
  });

  const isDead = (q: (typeof results)[number]) =>
    (q.error instanceof ApiError && q.error.status === 404) ||
    (!!q.data && FINISHED.includes(q.data.status));

  const signature = results.map((q) => `${q.status}:${q.data?.status ?? ''}`).join('|');
  useEffect(() => {
    stored.forEach((t, i) => {
      if (results[i] && isDead(results[i])) removeTicket(t.ticketId);
    });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [signature]);

  const tickets: ActiveTicket[] = [];
  let isChecking = false;
  stored.forEach((s, i) => {
    const q = results[i];
    if (!q || q.isPending) {
      isChecking = true;
      return;
    }
    if (isDead(q)) return;
    // A network error is not proof the ticket is gone, so keep showing it.
    tickets.push({ stored: s, ticket: q.data ?? null });
  });

  return { tickets, isChecking };
}

// Only ticket identifiers live here. Never a name or phone number (spec §20).
// frontend/src/lib/activeTickets.ts
const KEY = 'svq.tickets.v1';
const MAX_AGE_MS = 24 * 60 * 60 * 1000;
const MAX_TICKETS = 5;

export interface StoredTicket {
  ticketId: string;
  accessToken: string;
  salonId: string;
  tokenNumber: number;
  savedAt: number;
}

function isStored(x: unknown): x is StoredTicket {
  const t = x as Partial<StoredTicket> | null;
  return (
    !!t &&
    typeof t.ticketId === 'string' &&
    typeof t.accessToken === 'string' &&
    typeof t.salonId === 'string' &&
    typeof t.tokenNumber === 'number' &&
    typeof t.savedAt === 'number'
  );
}

// Every storage call is wrapped: private mode, blocked storage or a corrupt value
// must degrade to "no saved tickets", never crash the page.
function read(): StoredTicket[] {
  try {
    const raw = localStorage.getItem(KEY);
    if (!raw) return [];
    const parsed: unknown = JSON.parse(raw);
    if (!Array.isArray(parsed)) return [];
    const now = Date.now();
    return parsed.filter(isStored).filter((t) => now - t.savedAt < MAX_AGE_MS);
  } catch {
    return [];
  }
}

function write(list: StoredTicket[]) {
  try {
    localStorage.setItem(KEY, JSON.stringify(list.slice(-MAX_TICKETS)));
  } catch {
    /* storage unavailable: resume simply won't work on this browser */
  }
}

export function getTickets(salonId?: string): StoredTicket[] {
  const all = read();
  return salonId ? all.filter((t) => t.salonId === salonId) : all;
}

export function saveTicket(t: Omit<StoredTicket, 'savedAt'>) {
  write([...read().filter((x) => x.ticketId !== t.ticketId), { ...t, savedAt: Date.now() }]);
}

export function removeTicket(ticketId: string) {
  write(read().filter((x) => x.ticketId !== ticketId));
}

const PREFIX = 'joinKey:';
const MAX_AGE_MS = 6 * 60 * 60 * 1000;

function randomKey(): string {
  const bytes = new Uint8Array(16);
  crypto.getRandomValues(bytes);
  return Array.from(bytes, (b) => b.toString(16).padStart(2, '0')).join('');
}

// The same key for the same scan, even across Try again, reloads and tabs.
export function getJoinKey(salonId: string): string {
  try {
    const raw = localStorage.getItem(PREFIX + salonId);
    if (raw) {
      const { k, t } = JSON.parse(raw) as { k?: unknown; t?: unknown };
      if (typeof k === 'string' && typeof t === 'number' && Date.now() - t < MAX_AGE_MS) return k;
    }
  } catch { /* storage unavailable: fall through to a fresh key */ }
  const k = randomKey();
  try { localStorage.setItem(PREFIX + salonId, JSON.stringify({ k, t: Date.now() })); } catch { /* ignore */ }
  return k;
}

export function clearJoinKey(salonId: string) {
  try { localStorage.removeItem(PREFIX + salonId); } catch { /* ignore */ }
}

import { useState } from 'react';
import { useParams, useNavigate, Link } from 'react-router-dom';
import { joinSalonQueue } from '../api/tickets';
import { saveTicket } from '../lib/activeTickets';
import { useActiveTickets } from '../hooks/useActiveTickets';
import PageShell from '../components/PageShell';

export default function JoinQueue() {
  const { salonId } = useParams<{ salonId: string }>();
  const navigate = useNavigate();
  const { tickets: existing, isChecking } = useActiveTickets(salonId);
  const [addingAnother, setAddingAnother] = useState(false);
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleJoin() {
    setSubmitting(true);
    setError(null);
    try {
      const ticket = await joinSalonQueue(salonId!);
      saveTicket({
        ticketId: ticket.ticketId,
        accessToken: ticket.accessToken,
        salonId: ticket.salonId ?? salonId!,
        tokenNumber: ticket.tokenNumber,
      });
      navigate(`/tickets/${ticket.ticketId}?accessToken=${ticket.accessToken}`);
    } catch (err: any) {
      setError(err.message ?? 'Unable to join the queue. Please try again.');
    } finally {
      setSubmitting(false);
    }
  }

  const header = (
    <>
      <Link to={`/salons/${salonId}`} className="mb-3 inline-block min-h-11 py-2 text-sm text-gray-600 underline">
        ← Back to salon
      </Link>
      <h1 className="mb-1 text-xl font-semibold">Join the queue</h1>
    </>
  );

  if (isChecking) {
    return (
      <PageShell title="Join the queue">
        {header}
        <p role="status" className="text-gray-600">Checking for an existing ticket...</p>
      </PageShell>
    );
  }

  if (existing.length > 0 && !addingAnother) {
    const first = existing[0];
    return (
      <PageShell title="Join the queue">
        {header}
        <div role="alert" className="rounded-xl border border-amber-300 bg-amber-50 p-4 text-amber-900">
          <p className="font-semibold">You already have an active queue ticket.</p>
          <p className="mt-1 text-sm">
            Token #{first.stored.tokenNumber}
            {first.ticket && ` · ${first.ticket.peopleAhead} ${first.ticket.peopleAhead === 1 ? 'person' : 'people'} ahead`}
          </p>
        </div>
        <Link
          to={`/tickets/${first.stored.ticketId}?accessToken=${first.stored.accessToken}`}
          className="btn-primary mt-4"
        >
          Go to my ticket
        </Link>
        <button type="button" onClick={() => setAddingAnother(true)} className="btn-secondary mt-3">
          Join for another person
        </button>
      </PageShell>
    );
  }

  return (
    <PageShell title="Join the queue">
      {header}
      <p className="mb-4 text-sm text-gray-600">
        No account, name or phone number needed. Tap once to get your token.
      </p>
      {error && (
        <p id="join-error" role="alert" className="mb-4 rounded-lg border border-red-300 bg-red-50 p-3 text-sm text-red-900">
          {error}
        </p>
      )}
      <button type="button" onClick={handleJoin} disabled={submitting} className="btn-primary">
        {submitting ? 'Joining queue...' : 'Get my token'}
      </button>
    </PageShell>
  );
}

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';

export default function QueueTracking() {
  const { ticketId } = useParams<{ ticketId: string }>();
  const [searchParams] = useSearchParams();
  const accessToken = searchParams.get('accessToken') ?? '';
  const queryClient = useQueryClient();
  const queryKey = ['ticket', ticketId];
  const [confirmingLeave, setConfirmingLeave] = useState(false);

  const { data: ticket, isLoading, error } = useQuery({
    queryKey,
    queryFn: async () => {
      const t = await fetchTicketStatus(ticketId!, accessToken);
      return t ? { ...t, fetchedAt: Date.now() } : t;
    },
    enabled: !!ticketId && !!accessToken,
  });

  const leave = useMutation({
    mutationFn: () => leaveTicket(ticketId!, accessToken),
    onSuccess: () => {
      queryClient.setQueryData(queryKey, (prev: any) =>
        prev ? { ...prev, status: 'CANCELLED' } : prev,
      );
    },
    // e.g. the barber called this customer a moment ago: refetch so the
    // screen shows the real status instead of a stale Leave button.
    onError: () => {
      setConfirmingLeave(false);
      queryClient.invalidateQueries({ queryKey });
    },
  });

  // Join the salon's room once we know which salon this ticket belongs to.
  useEffect(() => {
    if (!ticket?.salonId) return;
    const salonId = ticket.salonId;

    function joinRoom() {
      socket.emit('join:salon', { salonId });
      queryClient.invalidateQueries({ queryKey }); // catch up after reconnects
    }

    joinRoom();
    socket.on('connect', joinRoom);

    return () => {
      socket.off('connect', joinRoom);
      socket.emit('leave:salon', { salonId });
    };
  }, [ticket?.salonId]);

  // Merge live broadcasts into this ticket's cached view.
  useEffect(() => {
    if (!ticketId) return;

    function handleUpdate(state: SanitizedQueueState) {
      const mine = state.entries.find((e) => e.id === ticketId);

      // Finished tickets drop out of the broadcast; refetch to learn which
      // final status we got (completed / skipped / cancelled).
      if (!mine) {
        queryClient.invalidateQueries({ queryKey });
        return;
      }

      queryClient.setQueryData(queryKey, (prev: any) => {
        if (!prev) return prev;
        const idx = state.entries.findIndex((e) => e.id === ticketId);
        const peopleAhead = state.entries
          .slice(0, idx)
          .filter((e) => e.status === 'WAITING').length;
        return {
          ...prev,
          status: mine.status,
          currentToken: state.currentToken,
          queueStatus: state.queueStatus,
          peopleAhead,
        };
      });
      // Wait estimate and countdown come from the server.
      queryClient.invalidateQueries({ queryKey });
    }

    socket.on('queue:updated', handleUpdate);
    return () => {
      socket.off('queue:updated', handleUpdate);
    };
  }, [ticketId, queryClient]);

  // Tab title as a cheap in-page alert for people who switched tabs.
  useEffect(() => {
    if (!ticket) return;
    const original = document.title;
    if (ticket.status === 'CALLED' || ticket.status === 'SERVING') document.title = `Your turn! #${ticket.tokenNumber}`;
    else if (ticket.status === 'WAITING' && ticket.peopleAhead <= 1)
      document.title = `Almost your turn · #${ticket.tokenNumber}`;
    return () => {
      document.title = original;
    };
  }, [ticket?.status, ticket?.peopleAhead, ticket?.tokenNumber]);

  // Keep the saved-ticket list honest: remember an active ticket (also covers opening the
  // link from another device), forget a finished or vanished one.
  useEffect(() => {
    if (!ticket || !ticketId) return;
    if (FINISHED.includes(ticket.status)) {
      removeTicket(ticketId);
    } else {
      saveTicket({ ticketId, accessToken, salonId: ticket.salonId, tokenNumber: ticket.tokenNumber });
    }
  }, [ticket?.status, ticketId]);

  useEffect(() => {
    if (error instanceof ApiError && error.status === 404 && ticketId) removeTicket(ticketId);
  }, [error, ticketId]);
  const homeLabel = CUSTOMER_DISCOVERY ? 'Find a salon' : 'Back to home';

  // ── Loading ────────────────────────────────────────────────────────────
  if (isLoading) {
    return (
      <PageShell title="Your queue ticket">
        <div className="animate-pulse space-y-4" aria-hidden="true">
          <div className="h-40 rounded-3xl bg-gray-200" />
          <div className="h-32 rounded-xl bg-gray-200" />
        </div>
        <p role="status" className="mt-4 text-center text-gray-600">Loading your queue ticket...</p>
      </PageShell>
    );
  }

  // ── Missing / invalid ticket ───────────────────────────────────────────
  if (error || !ticket) {
    return (
      <PageShell title="Your queue ticket">
        <div className="card p-6 text-center">
          <span aria-hidden="true" className="mb-3 block text-4xl">🔍</span>
          <p role="alert" className="text-red-700">
            We couldn't find this queue ticket. It may have expired, or the link is incomplete.
          </p>
          <Link to="/" className="btn-secondary mt-6">{homeLabel}</Link>
        </div>
      </PageShell>
    );
  }

  // ── Finished states ────────────────────────────────────────────────────
  if (FINISHED.includes(ticket.status)) {
    const message =
      ticket.status === 'COMPLETED'
        ? 'Your service is complete. Thank you!'
        : ticket.status === 'SKIPPED'
          ? 'The salon skipped your turn. If this looks like a mistake, please speak to the salon staff.'
          : 'You left the queue.';
    const icon = ticket.status === 'COMPLETED' ? '✅' : ticket.status === 'SKIPPED' ? '⏭️' : '👋';
    return (
      <PageShell title="Your queue ticket">
        <div className="card p-8 text-center">
          <span aria-hidden="true" className="mb-2 block text-4xl">{icon}</span>
          <p className="text-5xl font-bold tabular-nums text-gray-500">#{ticket.tokenNumber}</p>
          <p className="mt-4 text-gray-800" role="status">{message}</p>
          {ticket.status === 'COMPLETED' && ticket.prize && (
            <PrizeCard prize={ticket.prize} status={ticket.status} />
          )}
          <Link to="/" className="btn-secondary mt-6">{homeLabel}</Link>
        </div>
      </PageShell>
    );
  }

  // ── Status banner (color + icon + text, never color alone) ─────────────
  let banner: { icon: string; text: string; detail: string; classes: string } | null = null;
  if (ticket.status === 'CALLED') {
    banner = { icon: '🔔', text: "It's your turn", detail: 'Please head to the counter now.', classes: 'border-green-400 bg-green-50 text-green-900' };
  } else if (ticket.status === 'SERVING') {
    banner = { icon: '🔔', text: "It's your turn", detail: 'Please head to the counter now. Your service time is running.', classes: 'border-green-400 bg-green-50 text-green-900' };
  } else if (ticket.status === 'WAITING' && ticket.peopleAhead === 0) {
    banner = { icon: '🚶', text: "You're next", detail: 'Start making your way to the salon.', classes: 'border-amber-400 bg-amber-50 text-amber-900' };
  } else if (ticket.status === 'WAITING' && ticket.peopleAhead === 1) {
    banner = { icon: '⏳', text: "You're getting close", detail: 'Only one person is ahead of you.', classes: 'border-amber-300 bg-amber-50 text-amber-900' };
  }

  const canLeave = ticket.status === 'WAITING';
  const leaveError = leave.error instanceof Error ? leave.error.message : leave.error ? 'Unable to leave the queue. Please try again.' : null;

  return (
    <PageShell title="Your queue ticket">
      {/* Token */}
      <section aria-label="Your token" className="rounded-3xl border-2 border-ink bg-brand px-6 py-8 text-center text-ink shadow-[0_6px_0_0_var(--color-ink)]">
        <p className="text-sm font-bold uppercase tracking-widest">Your token</p>
        <p className="my-1 text-8xl font-extrabold tabular-nums">#{ticket.tokenNumber}</p>
        <div aria-hidden="true" className="mx-auto mt-4 border-t-2 border-dashed border-ink/40" />
      </section>

      {ticket.queueStatus === 'PAUSED' && (
        <p className="mt-4 flex items-start gap-2 rounded-xl border border-amber-300 bg-amber-50 p-3 text-sm text-amber-900" role="status">
          <span aria-hidden="true">⏸</span>
          <span>The salon has paused the queue for now. Your place is saved.</span>
        </p>
      )}
      {ticket.status === 'WAITING' && ticket.movedAhead && (
        <p className="mt-4 flex items-start gap-2 rounded-xl border border-blue-300 bg-blue-50 p-3 text-sm text-blue-900" role="status">
          <span aria-hidden="true">ℹ️</span>
          <span>A customer with an urgent need was moved ahead of you.</span>
        </p>
      )}
      {banner && (
        <div
          className={`mt-4 flex items-start gap-3 rounded-2xl border-2 p-4 ${banner.classes}`}
          role="status"
          aria-live="polite"
        >
          <span aria-hidden="true" className="text-3xl leading-none">{banner.icon}</span>
          <div>
            <p className="text-xl font-semibold">{banner.text}</p>
            <p className="text-sm">{banner.detail}</p>
          </div>
        </div>
      )}

      {ticket.status === 'SERVING' && ticket.remainingSeconds != null && (
        <section aria-label="Service time left" className="card mt-4 p-4 text-center">
          <p className="text-sm text-gray-600">Time left for your service</p>
          <Countdown
            endAtMs={ticket.fetchedAt + ticket.remainingSeconds * 1000}
            onExpired={() => queryClient.invalidateQueries({ queryKey })}
          />
        </section>
      )}

      {ticket.services?.length > 0 && (
        <section aria-label="Your services" className="card mt-4 p-4">
          <p className="mb-2 text-sm text-gray-600">Your services</p>
          <ul className="grid gap-1">
            {ticket.services.map((s: { name: string; durationMinutes: number }) => (
              <li key={s.name} className="flex justify-between gap-3">
                <span className="font-semibold">{s.name}</span>
                <span className="text-sm text-gray-700">{s.durationMinutes} min</span>
              </li>
            ))}
          </ul>
          <p className="mt-2 text-sm text-gray-700">Total time: ~{ticket.durationMinutes} min</p>
        </section>
      )}

      {/* Live numbers */}
      <section aria-label="Queue status" className="card mt-4 divide-y divide-gray-200 p-0">
        <div className="flex items-center justify-between px-4 py-3">
          <span className="text-sm text-gray-600">Now serving</span>
          <span className="rounded-lg bg-brand-soft px-2 text-2xl font-bold tabular-nums">#{ticket.currentToken}</span>
        </div>
        {ticket.status === 'WAITING' && (
          <>
            <div className="px-4 py-3">
              <p className="text-lg font-semibold">
                {ticket.peopleAhead} {ticket.peopleAhead === 1 ? 'person' : 'people'} ahead
              </p>
            </div>
            <div className="px-4 py-3">
              <p className="text-sm text-gray-700">Estimated wait: ~{ticket.estimatedWaitMinutes} min</p>
            </div>
          </>
        )}
      </section>
      {ticket.prize && <PrizeCard prize={ticket.prize} status={ticket.status} />}

      {ticket.status === 'WAITING' && (
        <NotifyMe ticketId={ticket.ticketId} accessToken={accessToken} enabled={!!ticket.notificationsOn} />
      )}

      {!ticket.notificationsOn ? (
        <p className="mt-4 text-center text-sm text-gray-600">
          Keep this page open. It updates live as the queue moves.
        </p>
      ) : (
        ticket.status === 'WAITING' && (
          <p className="mt-4 text-center text-sm text-gray-600">
            You can close this page. Keep your internet on and we'll alert you as your turn gets close.
          </p>
        )
      )}

      {canLeave && (
        <div className="mt-8">
          {!confirmingLeave ? (
            <button type="button" onClick={() => setConfirmingLeave(true)} className="btn-secondary">
              Leave queue
            </button>
          ) : (
            <div className="rounded-xl border border-red-200 bg-red-50 p-4" role="alertdialog" aria-label="Confirm leaving the queue">
              <p className="mb-3 text-sm text-red-900">Leave the queue? You will lose your place.</p>
              <div className="flex gap-3">
                <button
                  type="button"
                  onClick={() => leave.mutate()}
                  disabled={leave.isPending}
                  className="btn-danger flex-1"
                >
                  {leave.isPending ? 'Leaving...' : 'Yes, leave'}
                </button>
                <button
                  type="button"
                  onClick={() => setConfirmingLeave(false)}
                  disabled={leave.isPending}
                  className="btn-secondary flex-1"
                >
                  Stay
                </button>
              </div>
            </div>
          )}
          {leaveError && <p className="mt-3 text-sm text-red-700" role="alert">{leaveError}</p>}
        </div>
      )}
      {ticket.status === 'WAITING' && <LocalNews salonId={ticket.salonId} />}
    </PageShell>
  );
}

// frontend/src/pages/ScanLanding.tsx
// Where a salon's wall QR lands (/q/:code). Resume this phone's active ticket if it has one.
// Salons with services show a picker first; salons without join straight away.
import { useCallback, useEffect, useRef, useState } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import { fetchSalonByCode } from '../api/salons';
import { fetchSalonServices, type PublicService } from '../api/publicServices';
import { fetchTicketStatus, joinSalonQueue } from '../api/tickets';
import { ApiError } from '../api/client';
import { getTickets, removeTicket, saveTicket } from '../lib/activeTickets';
import { clearJoinKey, getJoinKey } from '../lib/joinKey';
import { askPushPermission, enablePushAfterJoin } from '../lib/push';
import PageShell from '../components/PageShell';
import ServicePicker from '../components/ServicePicker';

const FINISHED = ['COMPLETED', 'SKIPPED', 'CANCELLED'];

type View =
  | { kind: 'working'; salonName?: string }
  | { kind: 'choose'; salon: { id: string; name: string }; services: PublicService[]; error?: string }
  | { kind: 'error'; title: string; message: string; salonName?: string; canRetry: boolean };

const ticketUrl = (ticketId: string, accessToken: string) =>
  `/tickets/${ticketId}?accessToken=${accessToken}`;

// Two tabs (or a reload) scanning the same salon take turns, so the second one finds the first one's ticket.
async function withSalonLock<T>(salonId: string, fn: () => Promise<T>): Promise<T> {
  const locks = (navigator as unknown as { locks?: { request: <R>(name: string, cb: () => Promise<R>) => Promise<R> } }).locks;
  return locks?.request ? locks.request(`join:${salonId}`, fn) : fn();
}

export default function ScanLanding() {
  const { code } = useParams<{ code: string }>();
  const navigate = useNavigate();
  const [view, setView] = useState<View>({ kind: 'working' });
  const [joining, setJoining] = useState(false);
  // React StrictMode runs effects twice in development. This makes sure a scan can never take two tokens.
  const started = useRef(false);
  // A second tap on "Try again" while a request is in flight does nothing.
  const running = useRef(false);

  // The same key for every attempt of this scan, so the server never creates a second ticket for it.
  const joinNow = useCallback(
    async (salonId: string, serviceIds: string[]) => {
      const ticket = await joinSalonQueue(salonId, getJoinKey(salonId), serviceIds);
      saveTicket({
        ticketId: ticket.ticketId,
        accessToken: ticket.accessToken,
        salonId: ticket.salonId ?? salonId,
        tokenNumber: ticket.tokenNumber,
      });
      clearJoinKey(salonId);
      navigate(ticketUrl(ticket.ticketId, ticket.accessToken), { replace: true });
      return ticket;
    },
    [navigate],
  );

  const runOnce = useCallback(async () => {
    setView({ kind: 'working' });
    let salonName: string | undefined;
    try {
      const salon = await fetchSalonByCode(code ?? '');
      salonName = salon.name;
      setView({ kind: 'working', salonName });

      await withSalonLock(salon.id, async () => {
        // Scanning again must never hand out a second token: go back to the ticket this phone already holds.
        for (const stored of getTickets(salon.id)) {
          try {
            const t = await fetchTicketStatus(stored.ticketId, stored.accessToken);
            if (t && !FINISHED.includes(t.status)) {
              navigate(ticketUrl(stored.ticketId, stored.accessToken), { replace: true });
              return;
            }
            removeTicket(stored.ticketId);
          } catch (err) {
            if (err instanceof ApiError && err.status === 404) removeTicket(stored.ticketId);
            // Any other failure (e.g. offline): we can't prove the ticket is dead, so keep it and try the next one.
          }
        }

        const services = await fetchSalonServices(salon.id);
        if (services.length > 0) {
          setView({ kind: 'choose', salon: { id: salon.id, name: salon.name }, services });
          return; // the lock is released while the customer chooses
        }
        await joinNow(salon.id, []);
      });
    } catch (err) {
      if (!salonName && err instanceof ApiError && err.status === 404) {
        setView({
          kind: 'error',
          title: 'QR code not recognised',
          message: 'Please scan the QR code displayed at the salon again, or ask the salon staff for help.',
          canRetry: false,
        });
        return;
      }
      setView({
        kind: 'error',
        title: salonName ? "Can't get a token right now" : "Couldn't open the queue",
        message:
          err instanceof ApiError
            ? err.message
            : 'We could not reach the server. Please check your internet connection and try again.',
        salonName,
        canRetry: true,
      });
    }
  }, [code, navigate, joinNow]);

  const run = useCallback(async () => {
    if (running.current) return;
    running.current = true;
    try {
      await runOnce();
    } finally {
      running.current = false;
    }
  }, [runOnce]);

  useEffect(() => {
    if (started.current) return;
    started.current = true;
    void run();
  }, [run]);

  function onChoose(serviceIds: string[]) {
    if (view.kind !== 'choose' || joining) return;
    const alerts = askPushPermission(); // must start straight from the tap, before any await
    const { salon, services } = view;
    setJoining(true);

    void (async () => {
      try {
        const ticket = await withSalonLock(salon.id, () => joinNow(salon.id, serviceIds));
        void alerts
          .then((p) => {
            if (p === 'granted') return enablePushAfterJoin(ticket.ticketId, ticket.accessToken);
          })
          .catch(() => {}); // the Notify me button on the ticket is the fallback
      } catch (err) {
        const message =
          err instanceof ApiError
            ? err.message
            : 'We could not reach the server. Please check your internet connection and try again.';
        // A service may have been removed meanwhile: show the current list with the message.
        let fresh = services;
        try {
          fresh = await fetchSalonServices(salon.id);
        } catch {
          /* keep the old list */
        }
        setView({ kind: 'choose', salon, services: fresh, error: message });
      } finally {
        setJoining(false);
      }
    })();
  }

  if (view.kind === 'choose') {
    return (
      <PageShell title="Choose your service">
        <p className="mb-1 text-sm text-gray-600">{view.salon.name}</p>
        <h1 className="mb-4 text-xl font-semibold">Choose your service</h1>
        <ServicePicker services={view.services} busy={joining} error={view.error} onJoin={onChoose} />
      </PageShell>
    );
  }

  if (view.kind === 'error') {
    return (
      <PageShell title="Get your token">
        <div className="flex min-h-[60dvh] flex-col justify-center text-center">
          {view.salonName && <p className="mb-2 text-sm text-gray-600">{view.salonName}</p>}
          <h1 className="text-xl font-semibold">{view.title}</h1>
          <p role="alert" className="mt-3 rounded-lg border border-red-300 bg-red-50 p-3 text-sm text-red-900">
            {view.message}
          </p>
          {view.canRetry && (
            <button type="button" onClick={() => void run()} className="btn-primary mt-6">
              Try again
            </button>
          )}
        </div>
      </PageShell>
    );
  }

  return (
    <PageShell title="Get your token">
      <div className="flex min-h-[60dvh] flex-col justify-center text-center">
        {view.salonName && <p className="mb-2 text-sm text-gray-600">{view.salonName}</p>}
        <h1 className="text-xl font-semibold">Getting your token...</h1>
        <p role="status" className="mt-2 text-gray-600">
          Please wait a moment. No sign-up needed.
        </p>
      </div>
    </PageShell>
  );
}
