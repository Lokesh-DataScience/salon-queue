CREATE OR REPLACE FUNCTION public.barber_next(p_salon_id uuid)
 RETURNS TABLE(completed_entry_id uuid, called_entry_id uuid, called_token_number integer)
 LANGUAGE plpgsql
AS $function$
declare
  v_queue_id uuid;
  v_serving_id uuid;
  v_already_called_id uuid;
  v_completed_id uuid;
  v_called_id uuid;
  v_called_token int;
begin
  select id into v_queue_id
  from queues
  where salon_id = p_salon_id and date = current_date
  for update;

  if v_queue_id is null then
    raise exception 'NO_ACTIVE_QUEUE';
  end if;

  -- Refuse to call a new customer while one is already CALLED and
  -- hasn't been started, completed, or skipped yet.
  select id into v_already_called_id
  from queue_entries
  where queue_id = v_queue_id and status = 'CALLED'
  for update;

  if v_already_called_id is not null then
    raise exception 'ALREADY_CALLED';
  end if;

  -- Complete whoever is currently being served, if anyone.
  select id into v_serving_id
  from queue_entries
  where queue_id = v_queue_id and status = 'SERVING'
  for update;

  if v_serving_id is not null then
    update queue_entries
    set status = 'COMPLETED', completed_at = now()
    where id = v_serving_id;
    v_completed_id := v_serving_id;
  end if;

  -- Call the next waiting entry (does NOT start service yet).
  select id, token_number into v_called_id, v_called_token
  from queue_entries
  where queue_id = v_queue_id and status = 'WAITING'
  order by token_number asc
  for update
  limit 1;

  if v_called_id is not null then
    update queue_entries
    set status = 'CALLED', called_at = now()
    where id = v_called_id;

    update queues
    set current_token = v_called_token
    where id = v_queue_id;
  end if;

  return query select v_completed_id, v_called_id, v_called_token;
end;
$function$
;

CREATE OR REPLACE FUNCTION public.cleanup_customer_data(p_retention_hours integer)
 RETURNS TABLE(cancelled_stale integer, anonymized integer)
 LANGUAGE plpgsql
AS $function$
declare
  v_cancelled int;
  v_anon int;
begin
  -- 1. Entries stuck active on a past day's queue become CANCELLED.
  update queue_entries e
  set status = 'CANCELLED', cancelled_at = now()
  from queues q
  where e.queue_id = q.id
    and q.date < current_date
    and e.status in ('WAITING', 'CALLED', 'SERVING');
  get diagnostics v_cancelled = row_count;

  -- 2. Push subscriptions are useless once a ticket is finished, and the endpoint
  --    is an identifier, so drop them immediately (no retention window).
  update queue_entries
  set push_subscription = null
  where status in ('COMPLETED', 'SKIPPED', 'CANCELLED')
    and push_subscription is not null;

  -- 3. Null out personal data on finished entries past the window.
  update queue_entries
  set customer_name = null, customer_phone = null
  where status in ('COMPLETED', 'SKIPPED', 'CANCELLED')
    and (customer_name is not null or customer_phone is not null)
    and coalesce(completed_at, skipped_at, cancelled_at)
        < now() - make_interval(hours => p_retention_hours);
  get diagnostics v_anon = row_count;

  return query select v_cancelled, v_anon;
end;
$function$
;

CREATE OR REPLACE FUNCTION public.join_queue(p_salon_id uuid)
 RETURNS TABLE(entry_id uuid, queue_id uuid, token_number integer, access_token text, status text, joined_at timestamp with time zone)
 LANGUAGE plpgsql
AS $function$
DECLARE
  v_is_open boolean;
  v_is_queue_active boolean;
  v_opening text;
  v_closing text;
  v_queue_id uuid;
  v_next_token int;
  v_entry_id uuid;
  v_access_token text;
BEGIN
  SELECT is_open, is_queue_active, opening_time, closing_time
    INTO v_is_open, v_is_queue_active, v_opening, v_closing
  FROM salons WHERE id = p_salon_id FOR UPDATE;

  IF NOT FOUND THEN RAISE EXCEPTION 'SALON_NOT_FOUND'; END IF;
  IF NOT v_is_open OR NOT is_within_hours(v_opening, v_closing) THEN RAISE EXCEPTION 'SALON_CLOSED'; END IF;
  IF NOT v_is_queue_active THEN RAISE EXCEPTION 'QUEUE_NOT_ACTIVE'; END IF;

  SELECT id INTO v_queue_id
  FROM queues WHERE salon_id = p_salon_id AND date = CURRENT_DATE
  FOR UPDATE;

  IF NOT FOUND THEN
    INSERT INTO queues (salon_id, date, current_token, next_token, status)
    VALUES (p_salon_id, CURRENT_DATE, 0, 1, 'ACTIVE')
    RETURNING id INTO v_queue_id;
  END IF;

  IF EXISTS (SELECT 1 FROM queues q WHERE q.id = v_queue_id AND q.status <> 'ACTIVE') THEN
    RAISE EXCEPTION 'QUEUE_NOT_ACTIVE';
  END IF;

  SELECT next_token INTO v_next_token FROM queues WHERE id = v_queue_id FOR UPDATE;
  UPDATE queues SET next_token = next_token + 1, updated_at = now() WHERE id = v_queue_id;

  v_access_token := encode(gen_random_bytes(32), 'hex');

  INSERT INTO queue_entries (queue_id, token_number, status, joined_at, access_token)
  VALUES (v_queue_id, v_next_token, 'WAITING', now(), v_access_token)
  RETURNING id INTO v_entry_id;

  RETURN QUERY SELECT v_entry_id, v_queue_id, v_next_token, v_access_token, 'WAITING'::text, now();
END;
$function$;

CREATE OR REPLACE FUNCTION public.nearby_salons(p_lat double precision, p_lng double precision, p_radius_km double precision DEFAULT 5)
 RETURNS TABLE(id uuid, name text, description text, phone text, address text, latitude double precision, longitude double precision, opening_time text, closing_time text, average_service_minutes integer, is_open boolean, is_queue_active boolean, distance_km double precision, queue_size bigint, current_token integer, estimated_wait_minutes integer)
 LANGUAGE plpgsql
 STABLE
AS $function$
BEGIN
  RETURN QUERY
  SELECT
    s.id, s.name, s.description, s.phone, s.address,
    s.latitude, s.longitude, s.opening_time, s.closing_time,
    s.average_service_minutes,
    (s.is_open AND is_within_hours(s.opening_time, s.closing_time)) AS is_open,
    s.is_queue_active,
    (6371 * acos(LEAST(1.0, GREATEST(-1.0,
      cos(radians(p_lat)) * cos(radians(s.latitude)) *
      cos(radians(s.longitude) - radians(p_lng)) +
      sin(radians(p_lat)) * sin(radians(s.latitude))
    )))) AS distance_km,
    COALESCE(qe.waiting_count, 0) AS queue_size,
    COALESCE(q.current_token, 0) AS current_token,
    (COALESCE(qe.waiting_count, 0) * s.average_service_minutes)::int AS estimated_wait_minutes
  FROM salons s
  LEFT JOIN queues q ON q.salon_id = s.id AND q.date = CURRENT_DATE
  LEFT JOIN (
    SELECT queue_id, COUNT(*) AS waiting_count
    FROM queue_entries
    WHERE status = 'WAITING'
    GROUP BY queue_id
  ) qe ON qe.queue_id = q.id
  WHERE (6371 * acos(LEAST(1.0, GREATEST(-1.0,
      cos(radians(p_lat)) * cos(radians(s.latitude)) *
      cos(radians(s.longitude) - radians(p_lng)) +
      sin(radians(p_lat)) * sin(radians(s.latitude))
    )))) <= p_radius_km
  ORDER BY distance_km ASC;
END;
$function$
;

-- lane_waits gains a column, which needs a drop (join_queue only looks it up when it runs).
drop function if exists public.lane_waits(uuid);
create function public.lane_waits(p_salon_id uuid)
returns table (
  lane_id uuid, name text, is_default boolean, created_at timestamptz,
  waiting_count integer, wait_minutes integer, is_paused boolean
)
language sql
stable
as $$
  with cfg as (
    select coalesce(s.average_service_minutes, 15) as avg_min from salons s where s.id = p_salon_id
  ),
  today as (
    select q.id from queues q where q.salon_id = p_salon_id and q.date = current_date
  ),
  multi as (
    select salon_is_multi_lane(p_salon_id) as yes
  )
  select
    l.id, l.name, l.is_default, l.created_at,
    (count(e.id) filter (where e.status = 'WAITING'))::int,
    coalesce(sum(
      case
        when e.status = 'SERVING' and e.ends_at is not null
          then greatest(0, ceil(extract(epoch from (e.ends_at - clock_timestamp())) / 60))
        else coalesce(e.duration_minutes, (select avg_min from cfg))
      end
    ), 0)::int,
    (l.is_paused and (select yes from multi))
  from lanes l
  left join queue_entries e
    on e.lane_id = l.id
   and e.queue_id in (select id from today)
   and e.status in ('WAITING', 'CALLED', 'SERVING')
  where l.salon_id = p_salon_id and l.is_active
  group by l.id
$$;
revoke all on function public.lane_waits(uuid) from public, anon, authenticated;
grant execute on function public.lane_waits(uuid) to service_role;

CREATE OR REPLACE FUNCTION public.join_queue(p_salon_id uuid, p_client_key text DEFAULT NULL, p_lane_id uuid DEFAULT NULL)
 RETURNS TABLE(entry_id uuid, queue_id uuid, token_number integer, access_token text, status text, joined_at timestamp with time zone)
 LANGUAGE plpgsql
AS $function$
DECLARE
  v_is_open boolean;
  v_is_queue_active boolean;
  v_opening text;
  v_closing text;
  v_queue_id uuid;
  v_next_token int;
  v_entry_id uuid;
  v_access_token text;
  v_expected int;
  v_scans int;
  v_left int;
  v_total float8;
  v_roll float8;
  v_cum float8;
  v_rec record;
  v_offer_id uuid;
  v_code text;
  v_lane_id uuid;
  v_lane_active boolean;
  v_lane_paused boolean;
BEGIN
  SELECT is_open, is_queue_active, opening_time, closing_time
    INTO v_is_open, v_is_queue_active, v_opening, v_closing
  FROM salons WHERE id = p_salon_id FOR UPDATE;

  IF NOT FOUND THEN RAISE EXCEPTION 'SALON_NOT_FOUND'; END IF;

  -- Same key again (double tap, retry, second tab): hand back the ticket that already exists.
  -- The salon row is locked above, so two simultaneous requests with one key cannot both pass this.
  IF p_client_key IS NOT NULL THEN
    RETURN QUERY
      SELECT e.id, e.queue_id, e.token_number, e.access_token, e.status::text, e.joined_at
      FROM queue_entries e JOIN queues q ON q.id = e.queue_id
      WHERE q.salon_id = p_salon_id AND q.date = CURRENT_DATE AND e.client_key = p_client_key
      LIMIT 1;
    IF FOUND THEN RETURN; END IF;
  END IF;

  IF NOT v_is_open OR NOT is_within_hours(v_opening, v_closing) THEN RAISE EXCEPTION 'SALON_CLOSED'; END IF;
  IF NOT v_is_queue_active THEN RAISE EXCEPTION 'QUEUE_NOT_ACTIVE'; END IF;

  SELECT id INTO v_queue_id
  FROM queues WHERE salon_id = p_salon_id AND date = CURRENT_DATE
  FOR UPDATE;

  IF NOT FOUND THEN
    INSERT INTO queues (salon_id, date, current_token, next_token, status)
    VALUES (p_salon_id, CURRENT_DATE, 0, 1, 'ACTIVE')
    RETURNING id INTO v_queue_id;
  END IF;

  IF EXISTS (SELECT 1 FROM queues q WHERE q.id = v_queue_id AND q.status <> 'ACTIVE') THEN
    RAISE EXCEPTION 'QUEUE_NOT_ACTIVE';
  END IF;

  SELECT next_token INTO v_next_token FROM queues WHERE id = v_queue_id FOR UPDATE;
  UPDATE queues SET next_token = next_token + 1, updated_at = now() WHERE id = v_queue_id;

  -- ── Which barber's line? (the salon row is locked, so the lanes cannot change under us) ──
  IF p_lane_id IS NOT NULL THEN
    SELECT l.id, l.is_active, l.is_paused INTO v_lane_id, v_lane_active, v_lane_paused
    FROM lanes l WHERE l.id = p_lane_id AND l.salon_id = p_salon_id;   -- another salon's lane looks like no lane
    IF NOT FOUND THEN RAISE EXCEPTION 'LANE_NOT_FOUND'; END IF;
    IF NOT v_lane_active THEN RAISE EXCEPTION 'LANE_INACTIVE'; END IF;
    -- A barber's break only counts when there is more than one barber (a one-barber salon pauses the whole queue).
    IF v_lane_paused AND salon_is_multi_lane(p_salon_id) THEN RAISE EXCEPTION 'LANE_PAUSED'; END IF;
  ELSE
    -- "Any barber": the line with the shortest wait; ties go to the fewest waiting, then the owner's own line.
    SELECT w.lane_id INTO v_lane_id
    FROM lane_waits(p_salon_id) w
    WHERE NOT w.is_paused                       -- never a barber who is on a break
    ORDER BY w.wait_minutes, w.waiting_count, w.is_default DESC, w.created_at, w.lane_id
    LIMIT 1;
    IF v_lane_id IS NULL THEN
      -- every barber is on a break: nobody to join
      IF EXISTS (SELECT 1 FROM lane_waits(p_salon_id) w2 WHERE w2.is_paused) THEN
        RAISE EXCEPTION 'QUEUE_NOT_ACTIVE';
      END IF;
      SELECT l.id INTO v_lane_id FROM lanes l WHERE l.salon_id = p_salon_id AND l.is_default;
    END IF;
  END IF;

  v_access_token := encode(gen_random_bytes(32), 'hex');

  INSERT INTO queue_entries (queue_id, token_number, status, joined_at, access_token, client_key, lane_id)
  VALUES (v_queue_id, v_next_token, 'WAITING', now(), v_access_token, p_client_key, v_lane_id)
  RETURNING id INTO v_entry_id;

  -- ── Reward draw: QR scans only, under the queue lock ──
  IF EXISTS (SELECT 1 FROM salon_offers o WHERE o.salon_id = p_salon_id AND o.active AND o.deleted_at IS NULL) THEN

    SELECT s.expected_daily_customers INTO v_expected FROM salons s WHERE s.id = p_salon_id;
    IF v_expected IS NULL THEN
      SELECT q.prize_expected INTO v_expected FROM queues q WHERE q.id = v_queue_id;
      IF v_expected IS NULL THEN
        SELECT round(avg(t.c))::int INTO v_expected FROM (
          SELECT count(*) AS c
          FROM queue_entries e JOIN queues q2 ON q2.id = e.queue_id
          WHERE q2.salon_id = p_salon_id
            AND q2.date < CURRENT_DATE AND q2.date >= CURRENT_DATE - 7
            AND e.status <> 'CANCELLED'
            AND NOT coalesce(e.is_walk_in, false)
          GROUP BY q2.id
        ) t;
        v_expected := greatest(coalesce(v_expected, 15), 15);
        UPDATE queues q SET prize_expected = v_expected WHERE q.id = v_queue_id;
      END IF;
    END IF;

    SELECT count(*) INTO v_scans
    FROM queue_entries e
    WHERE e.queue_id = v_queue_id AND NOT coalesce(e.is_walk_in, false);

    -- Only the first `expected` QR customers are eligible.
    IF v_scans <= v_expected THEN
      v_left := v_expected - v_scans + 1;

      SELECT coalesce(sum(least(1.0, r.rem::float8 / v_left)), 0) INTO v_total
      FROM (
        SELECT greatest(0, o.daily_count - (
                 SELECT count(*) FROM queue_entries e WHERE e.queue_id = v_queue_id AND e.offer_id = o.id
               )) AS rem
        FROM salon_offers o
        WHERE o.salon_id = p_salon_id AND o.active AND o.deleted_at IS NULL
      ) r
      WHERE r.rem > 0;

      IF v_total > 0 THEN
        v_roll := random() * greatest(1.0, v_total);
        v_cum := 0;
        FOR v_rec IN
          SELECT r.oid, r.rem FROM (
            SELECT o.id AS oid,
                   greatest(0, o.daily_count - (
                     SELECT count(*) FROM queue_entries e WHERE e.queue_id = v_queue_id AND e.offer_id = o.id
                   )) AS rem
            FROM salon_offers o
            WHERE o.salon_id = p_salon_id AND o.active AND o.deleted_at IS NULL
          ) r
          WHERE r.rem > 0
          ORDER BY r.oid
        LOOP
          v_cum := v_cum + least(1.0, v_rec.rem::float8 / v_left);
          IF v_roll < v_cum THEN v_offer_id := v_rec.oid; EXIT; END IF;
        END LOOP;

        IF v_offer_id IS NOT NULL THEN
          LOOP
            v_code := (
              SELECT string_agg(substr('ABCDEFGHJKLMNPQRSTUVWXYZ23456789', (get_byte(gen_random_bytes(1), 0) % 32) + 1, 1), '')
              FROM generate_series(1, 6)
            );
            EXIT WHEN NOT EXISTS (
              SELECT 1 FROM queue_entries e WHERE e.queue_id = v_queue_id AND e.prize_code = v_code
            );
          END LOOP;

          UPDATE queue_entries e SET offer_id = v_offer_id, prize_code = v_code WHERE e.id = v_entry_id;
        END IF;
      END IF;
    END IF;
  END IF;

  RETURN QUERY SELECT v_entry_id, v_queue_id, v_next_token, v_access_token, 'WAITING'::text, now();
END;
$function$;



REVOKE ALL ON FUNCTION public.join_queue(uuid, text, uuid) FROM public, anon, authenticated;
GRANT EXECUTE ON FUNCTION public.join_queue(uuid, text, uuid) TO service_role;

drop function if exists public.add_walk_in(uuid);
create or replace function public.add_walk_in(p_salon_id uuid, p_lane_id uuid default null)
 returns table(entry_id uuid, token_number integer)
 language plpgsql
as $function$
declare
  v_is_open boolean;
  v_is_queue_active boolean;
  v_opening text;
  v_closing text;
  v_queue_id uuid;
  v_next_token int;
  v_entry_id uuid;
  v_lane_id uuid;
  v_lane_active boolean;
  v_lane_paused boolean;
begin
  select s.is_open, s.is_queue_active, s.opening_time, s.closing_time
    into v_is_open, v_is_queue_active, v_opening, v_closing
  from salons s where s.id = p_salon_id for update;

  if not found then raise exception 'SALON_NOT_FOUND'; end if;
  if not v_is_open or not is_within_hours(v_opening, v_closing) then raise exception 'SALON_CLOSED'; end if;
  if not v_is_queue_active then raise exception 'QUEUE_NOT_ACTIVE'; end if;

  select q.id into v_queue_id
  from queues q where q.salon_id = p_salon_id and q.date = current_date
  for update;

  if not found then
    insert into queues (salon_id, date, current_token, next_token, status)
    values (p_salon_id, current_date, 0, 1, 'ACTIVE')
    returning id into v_queue_id;
  end if;

  if exists (select 1 from queues q where q.id = v_queue_id and q.status <> 'ACTIVE') then
    raise exception 'QUEUE_NOT_ACTIVE';
  end if;

  -- Which barber's line? No lane given = the owner's own (default) line.
  if p_lane_id is null then
    select l.id into v_lane_id from lanes l where l.salon_id = p_salon_id and l.is_default;
  else
    select l.id, l.is_active, l.is_paused into v_lane_id, v_lane_active, v_lane_paused
    from lanes l where l.id = p_lane_id and l.salon_id = p_salon_id;
    if not found then raise exception 'LANE_NOT_FOUND'; end if;
    if not v_lane_active then raise exception 'LANE_INACTIVE'; end if;
    if v_lane_paused and salon_is_multi_lane(p_salon_id) then raise exception 'LANE_PAUSED'; end if;
  end if;

  select q.next_token into v_next_token from queues q where q.id = v_queue_id for update;
  update queues q set next_token = q.next_token + 1, updated_at = now() where q.id = v_queue_id;

  -- Same shape as a normal ticket, but flagged as a walk-in and with NO prize draw
  -- (a walk-in has no phone to reveal a prize on).
  insert into queue_entries (queue_id, token_number, status, joined_at, access_token, is_walk_in, lane_id)
  values (v_queue_id, v_next_token, 'WAITING', now(), encode(gen_random_bytes(32), 'hex'), true, v_lane_id)
  returning id into v_entry_id;

  return query select v_entry_id, v_next_token;
end;
$function$;


revoke all on function public.add_walk_in(uuid, uuid) from public, anon, authenticated;
grant execute on function public.add_walk_in(uuid, uuid) to service_role;

commit;

notify pgrst, 'reload schema';

import bcrypt from 'bcrypt';
import { supabase } from '../config/supabase';
import { AppError } from '../middleware/errorHandler';

const MAX_BARBERS = 10;

interface TeamRow {
  id: string;
  name: string;
  phone: string | null;
  email: string | null;
  role: 'OWNER' | 'BARBER' | 'MANAGER';
  is_active: boolean;
  lane_id: string | null;
}

const toDto = (r: TeamRow) => ({
  id: r.id,
  name: r.name,
  phone: r.phone,
  email: r.email,
  role: r.role,
  isActive: r.is_active,
  laneId: r.lane_id,
});

// Every function takes salonId from the verified session, never from the request.
export async function listTeam(salonId: string) {
  const { data, error } = await supabase
    .from('salon_staff')
    .select('id, name, phone, email, role, is_active, lane_id')
    .eq('salon_id', salonId)
    .order('created_at', { ascending: true })
    .order('id', { ascending: true });

  if (error) throw error;
  return ((data ?? []) as TeamRow[]).map(toDto);
}

// Creates the barber and their lane together (one transaction in Postgres).
export async function addBarber(
  salonId: string,
  input: { name: string; phone: string; password: string },
) {
  const passwordHash = await bcrypt.hash(input.password, 12);

  const { data, error } = await supabase.rpc('add_staff_with_lane', {
    p_salon_id: salonId,
    p_name: input.name,
    p_phone: input.phone,
    p_password_hash: passwordHash,
  });

  if (error) {
    const msg = error.message ?? '';
    if (msg.includes('PHONE_TAKEN')) throw new AppError('That phone number is already registered.', 409);
    if (msg.includes('TOO_MANY_BARBERS')) throw new AppError(`You can have at most ${MAX_BARBERS} barbers.`, 409);
    if (msg.includes('SALON_NOT_FOUND')) throw new AppError('Salon not found.', 404);
    throw error;
  }

  const row = Array.isArray(data) ? data[0] : data;
  if (!row?.o_staff_id) throw new AppError('Could not add the barber. Please try again.', 500);

  return toDto({
    id: row.o_staff_id as string,
    name: input.name,
    phone: input.phone,
    email: null,
    role: 'BARBER',
    is_active: true,
    lane_id: row.o_lane_id as string,
  });
}

// The owner sets a new password for a barber who forgot theirs.
export async function resetBarberPassword(salonId: string, staffId: string, password: string) {
  const passwordHash = await bcrypt.hash(password, 12);

  const { data, error } = await supabase
    .from('salon_staff')
    .update({ password_hash: passwordHash })
    .eq('id', staffId)
    .eq('salon_id', salonId) // ownership: always the session's salon
    .neq('role', 'OWNER')
    .select('id')
    .maybeSingle();

  if (error) throw error;
  if (!data) throw new AppError('Barber not found.', 404);
}

export async function setBarberActive(salonId: string, staffId: string, isActive: boolean) {
  const { error } = await supabase.rpc('set_staff_active', {
    p_salon_id: salonId,
    p_staff_id: staffId,
    p_active: isActive,
  });

  if (error) {
    const msg = error.message ?? '';
    if (msg.includes('STAFF_NOT_FOUND')) throw new AppError('Barber not found.', 404);
    if (msg.includes('LANE_HAS_CUSTOMERS')) {
      throw new AppError(
        'This barber still has customers in their line. Switch them off once the line is empty.',
        409,
      );
    }
    if (msg.includes('TOO_MANY_BARBERS')) throw new AppError(`You can have at most ${MAX_BARBERS} barbers.`, 409);
    throw error;
  }
  return { id: staffId, isActive };
}

// Who am I? Read from the database (not the token) so a switched-off account stops working
// on its very next page load instead of lingering until the token expires.
export async function getStaffMe(staffId: string) {
  const { data, error } = await supabase
    .from('salon_staff')
    .select('id, name, role, salon_id, lane_id, is_active, lanes(name)')
    .eq('id', staffId)
    .maybeSingle();

  if (error) throw error;
  if (!data || !data.is_active) return null;

  const lane = Array.isArray(data.lanes) ? data.lanes[0] : data.lanes;
  return {
    id: data.id as string,
    name: data.name as string,
    role: data.role as 'OWNER' | 'BARBER' | 'MANAGER',
    salonId: data.salon_id as string,
    laneId: (data.lane_id as string | null) ?? null,
    laneName: (lane as { name: string } | null | undefined)?.name ?? null,
  };
}

// Any signed-in staff member changes their own password. A wrong current password is a 400, not
// a 401: the app reads 401 as "your session ended" and would log the person out.
export async function changeOwnPassword(staffId: string, currentPassword: string, newPassword: string) {
  const { data, error } = await supabase
    .from('salon_staff')
    .select('id, password_hash, is_active')
    .eq('id', staffId)
    .maybeSingle();
  if (error) throw error;
  if (!data || !data.is_active) throw new AppError('Session expired. Please log in again.', 401);

  const matches = await bcrypt.compare(currentPassword, data.password_hash as string);
  if (!matches) throw new AppError('Your current password is not correct.', 400);
  if (currentPassword === newPassword) throw new AppError('Choose a password you have not used before.', 400);

  const passwordHash = await bcrypt.hash(newPassword, 12);
  const { error: updateError } = await supabase
    .from('salon_staff')
    .update({ password_hash: passwordHash })
    .eq('id', staffId);
  if (updateError) throw updateError;
}

export async function getSalonQueueSnapshot(salonId: string) {
  const { data: queue } = await supabase
    .from('queues')
    .select('*')
    .eq('salon_id', salonId)
    .eq('date', todayISODate())
    .maybeSingle();

  if (!queue) {
    return { currentToken: 0, waitingCount: 0 };
  }

  const { count } = await supabase
    .from('queue_entries')
    .select('*', { count: 'exact', head: true })
    .eq('queue_id', queue.id)
    .eq('status', 'WAITING');

  return { currentToken: queue.current_token, waitingCount: count ?? 0 };
}

// Prefer the specific reason ("Enter a valid 10-digit mobile number.") over the generic "Invalid request."
function errText(e: unknown, fallback: string) {
  if (e instanceof ApiError && e.fieldErrors) {
    const first = Object.values(e.fieldErrors).flat().find(Boolean);
    if (first) return first;
  }
  return e instanceof Error ? e.message : fallback;
}
