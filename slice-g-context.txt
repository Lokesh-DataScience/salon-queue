// frontend/src/api/client.ts
// The live site is served by the same server as the API, so production always uses its own address.
const API_URL = import.meta.env.DEV ? (import.meta.env.VITE_API_URL ?? 'http://localhost:4000') : '';

export const SESSION_EXPIRED_EVENT = 'barber:session-expired';

export class ApiError extends Error {
  status: number;
  fieldErrors?: Record<string, string[] | undefined>;
  constructor(message: string, status: number, fieldErrors?: Record<string, string[] | undefined>) {
    super(message);
    this.status = status;
    this.fieldErrors = fieldErrors;
  }
}

type RefreshOutcome = 'ok' | 'denied' | 'error';
let refreshInFlight: Promise<RefreshOutcome> | null = null;

// Single-flight: ten requests failing at once trigger one refresh, not ten.
function refreshSession(): Promise<RefreshOutcome> {
  refreshInFlight ??= fetch(`${API_URL}/api/auth/barber/refresh`, {
    method: 'POST',
    credentials: 'include',
  })
    .then((res): RefreshOutcome => (res.ok ? 'ok' : res.status === 401 ? 'denied' : 'error'))
    .catch((): RefreshOutcome => 'error')
    .finally(() => {
      refreshInFlight = null;
    });
  return refreshInFlight;
}

export async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const { headers: extraHeaders, ...rest } = init ?? {};
  const send = () =>
    fetch(`${API_URL}${path}`, {
      credentials: 'include',
      ...rest,
      headers: { 'Content-Type': 'application/json', ...(extraHeaders as Record<string, string> | undefined) },
    });

  let res = await send();

  // Barber-only routes: if the access token expired, refresh silently and retry once.
  // (A 401 means the handler never ran, so retrying a POST cannot double-apply it.)
  if (res.status === 401 && path.startsWith('/api/barber/')) {
    const outcome = await refreshSession();
    // 'denied' can simply mean another tab rotated the token first. Its new cookies are
    // already in the shared cookie jar, so retry once for 'ok' and 'denied' alike.
    if (outcome !== 'error') {
      res = await send();
      if (res.status === 401) window.dispatchEvent(new Event(SESSION_EXPIRED_EVENT));
    }
  }

  if (!res.ok) {
    const body = await res.json().catch(() => ({ message: 'Something went wrong.' }));
    throw new ApiError(body.message ?? `Request failed with status ${res.status}`, res.status, body.errors);
  }

  return res.json() as Promise<T>;
}

  const navigate = useNavigate();
  const queryClient = useQueryClient();

  // Auth gate for every barber screen. Only a 401 sends the barber to login;
  // a network blip should not log anyone out.
  const { error } = useQuery({ queryKey: ['barberSalon'], queryFn: fetchBarberSalon, retry: false });

  useEffect(() => {
    function onExpired() {
      queryClient.clear();

// frontend/src/pages/BarberDashboard.tsx
import { useEffect, useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { useNavigate } from 'react-router-dom';
import {
  fetchBarberQueue,
  callNextCustomer,
  startEntry,
  completeEntry,
  skipEntry,
  pauseQueue,
  resumeQueue,
  addWalkIn,
  prioritizeEntry,
} from '../api/barber';
import { redeemCode } from '../api/offers';
import { socket } from '../lib/socket';
import { SESSION_EXPIRED_EVENT, ApiError } from '../api/client';

const QUEUE_KEY = ['barberQueue'];

export default function BarberDashboard() {
  const navigate = useNavigate();
  const queryClient = useQueryClient();
  const [actionError, setActionError] = useState<string | null>(null);
  const [redeemInput, setRedeemInput] = useState('');
  const [redeemMsg, setRedeemMsg] = useState<{ ok: boolean; text: string } | null>(null);
  const redeem = useMutation({
    mutationFn: () => redeemCode(redeemInput),
    onSuccess: (r) => { setRedeemMsg({ ok: true, text: `Redeemed: ${r.prizeName}. Give it to the customer.` }); setRedeemInput(''); },
    onError: (e: any) => setRedeemMsg({ ok: false, text: e.message ?? 'Could not redeem.' }),
  });

  const { data: queue, isLoading, error } = useQuery({
    queryKey: QUEUE_KEY,
    queryFn: fetchBarberQueue,
    retry: (count, err) => !(err instanceof ApiError && err.status === 401) && count < 2,
  });
  function refetch() {
    return queryClient.invalidateQueries({ queryKey: QUEUE_KEY });
  }

  // The API client emits this only after a silent refresh has failed.
  useEffect(() => {
    function onExpired() {
      queryClient.clear();
      navigate('/barber/login', { replace: true, state: { expired: true } });
    }
    window.addEventListener(SESSION_EXPIRED_EVENT, onExpired);
    return () => window.removeEventListener(SESSION_EXPIRED_EVENT, onExpired);
  }, [navigate, queryClient]);

  // queue:updated is only a signal to re-fetch the full snapshot; the broadcast
  // payload is sanitized and deliberately carries no customer details.
  useEffect(() => {
    if (!queue?.salonId) return;
    const salonId = queue.salonId;

    function joinRoom() {
      socket.emit('join:salon', { salonId });
      refetch();
    }

    joinRoom();
    socket.on('connect', joinRoom);
    socket.on('queue:updated', refetch);

    return () => {
      socket.off('connect', joinRoom);
      socket.off('queue:updated', refetch);
      socket.emit('leave:salon', { salonId });
    };
  }, [queue?.salonId]);

  function runAction<T>(action: () => Promise<T>) {
    setActionError(null);
    return action()
      .then(() => refetch())
      .catch((err: any) => {
        setActionError(err.message);
        return refetch(); // re-sync so the button reflects what really happened
      });
  }

  // ── The one primary action ─────────────────────────────────────────────
  // The server still enforces every transition; this only picks which call to make.
  const primaryMutation = useMutation({
    mutationFn: () =>
      runAction(async () => {
        if (!queue) return;
        if (queue.serving) {
          await completeEntry(queue.serving.id); // SERVING → COMPLETED
        } else if (queue.called) {
          await startEntry(queue.called.id); // CALLED → SERVING
        } else if (queue.queueStatus === 'PAUSED') {
          await resumeQueue();
        } else {
          // Idle: call the next customer and start them in one tap.
          // If start fails, the customer stays CALLED and the button becomes "Start #N".
          const { result } = await callNextCustomer();
          const calledId = (result as { called_entry_id?: string | null } | null)?.called_entry_id;
          if (calledId) await startEntry(calledId);
        }
      }),
  });
  const skipMutation = useMutation({ mutationFn: (id: string) => runAction(() => skipEntry(id)) });
  const pauseMutation = useMutation({ mutationFn: () => runAction(pauseQueue) });

  // Emergency: serve a waiting customer right after the current one.
  const [bumpTarget, setBumpTarget] = useState<{ id: string; token_number: number } | null>(null);
  const bumpMutation = useMutation({
    mutationFn: (id: string) => runAction(() => prioritizeEntry(id)),
    onSettled: () => setBumpTarget(null),
  });

  const [walkInMsg, setWalkInMsg] = useState<string | null>(null);
  useEffect(() => {
    if (!walkInMsg) return;
    const t = setTimeout(() => setWalkInMsg(null), 6000);
    return () => clearTimeout(t);
  }, [walkInMsg]);
  const walkInMutation = useMutation({
    mutationFn: addWalkIn,
    onMutate: () => {
      setActionError(null);
      setWalkInMsg(null);
    },
    onSuccess: (r) => {
      setWalkInMsg(`Token #${r.entry.tokenNumber} added to the queue.`);
      refetch();
    },
    onError: (err: any) => {
      setActionError(err.message);
      refetch();
    },
  });

  if (isLoading) {
    return (
      <main className="mx-auto w-full max-w-md p-4 md:max-w-2xl">
        <div className="animate-pulse space-y-4" aria-hidden="true">
          <div className="h-20 rounded-2xl bg-gray-200" />
          <div className="h-44 rounded-2xl bg-gray-200" />
          <div className="h-20 rounded-2xl bg-gray-200" />
        </div>
        <p role="status" className="mt-4 text-center text-gray-600">Loading queue...</p>
      </main>
    );
  }

  if (error || !queue) {
    return (
      <main className="mx-auto max-w-md p-6">
        <div className="card p-6 text-center">
          <p role="alert" className="mb-4 text-red-700">Not signed in, or your session expired.</p>
          <button onClick={() => navigate('/barber/login')} className="btn-primary">
            Go to Login
          </button>
        </div>
      </main>
    );
  }

  const isPaused = queue.queueStatus === 'PAUSED';
  const busy =
    primaryMutation.isPending ||
    skipMutation.isPending ||
    pauseMutation.isPending ||
    walkInMutation.isPending ||
    bumpMutation.isPending;
  const nextWaiting = queue.waiting[0] ?? null;
  const active = queue.serving ?? queue.called; // whoever is at the chair / being called

  // What the one button says and does right now.
  let label: string;
  let color = 'bg-brand text-ink hover:bg-brand-dark';
  let disabled = busy;
  if (queue.serving) {
    label = `COMPLETE #${queue.serving.token_number}`;
    color = 'bg-green-700 text-white hover:bg-green-800';
  } else if (queue.called) {
    label = `START #${queue.called.token_number}`;
  } else if (isPaused) {
    label = 'RESUME QUEUE';
    color = 'bg-amber-700 text-white hover:bg-amber-800';
  } else if (nextWaiting) {
    label = `START NEXT #${nextWaiting.token_number}`;
  } else {
    label = 'NO ONE WAITING';
    disabled = true;
  }
  if (primaryMutation.isPending) label = 'Working...';

  const statusLine = queue.serving
    ? 'Serving now'
    : queue.called
      ? 'Called — waiting to start'
      : isPaused
        ? 'On a break'
        : 'Ready';

  const cardTone = queue.serving
    ? 'border-green-600 bg-green-50'
    : queue.called
      ? 'border-ink bg-brand-soft'
      : 'border-gray-300 bg-white';

  return (
    <>
      <main className="mx-auto w-full max-w-md p-4 pb-36 md:max-w-2xl">
        <header className="mb-3 flex items-start justify-between gap-3 rounded-2xl bg-ink p-4 text-white">
          <div className="flex items-center gap-3">
            <img src="/icon-192.png" alt="" width={44} height={44} className="h-11 w-11 rounded-xl" />
            <div>
              <h1 className="text-xl font-semibold">Salon queue</h1>
              <p className="text-sm text-gray-300">Current token #{queue.currentToken}</p>
            </div>
          </div>
          <span
            className={`pill shrink-0 ${
              isPaused ? 'border-amber-300 bg-amber-50 text-amber-900' : 'border-green-300 bg-green-50 text-green-900'
            }`}
          >
            <span
              aria-hidden="true"
              className={`h-2 w-2 rounded-full ${isPaused ? 'bg-amber-600' : 'bg-green-600'}`}
            />
            {isPaused ? 'Queue paused' : 'Queue active'}
          </span>
        </header>

        {actionError && (
          <p role="alert" className="mb-4 rounded-lg border border-red-300 bg-red-50 p-3 text-sm text-red-900">
            {actionError}
          </p>
        )}

        <section
          aria-label="Current customer"

        >
          <p className="text-sm font-bold uppercase tracking-wide text-gray-700">{statusLine}</p>
          <p className="my-2 text-8xl font-extrabold tabular-nums text-ink">{active ? `#${active.token_number}` : '—'}</p>
          {!active && nextWaiting && !isPaused && (
            <p className="text-sm text-gray-700">Next up: #{nextWaiting.token_number}</p>
          )}
        </section>

        {/* Rare actions stay out of the way. Skip is here so a no-show can never block the queue. */}
        <details className="mb-6 rounded-xl border-2 border-gray-300 bg-white px-3">

      )}
    </>
  );
}

                  </div>
                ) : confirmDeleteId === o.id ? (

                      <button type="button" className="btn-secondary flex-1" onClick={() => setConfirmDeleteId(null)}>Keep</button>
                    </div>
                  </div>
                ) : (
                  <div className="mt-3 flex flex-wrap gap-2">
                    <button type="button" className="btn-secondary !w-auto"
                      onClick={() => patch.mutate({ id: o.id, p: { active: !o.active } })} disabled={patch.isPending}>

                    </button>
                    <button type="button" className="btn-secondary !w-auto"
                      onClick={() => { setEditingId(o.id); setEditCount(String(o.dailyCount)); }}>Edit</button>
                    <button type="button" className="btn-secondary !w-auto" onClick={() => setConfirmDeleteId(o.id)}>Delete</button>
                  </div>
                )}
              </li>
            ))}
          </ul>
        )}

type FieldErrors = Record<string, string[] | undefined>;
const AREA_FIELDS = ['district', 'state'] as const;

export default function BarberRegister() {
  const navigate = useNavigate();
  const [f, setF] = useState({
    salonName: '', phone: '', address: '', latitude: '', longitude: '',
    district: '', state: '',
    ownerName: '', email: '', password: '',
    openingTime: '09:00', closingTime: '21:00', averageServiceMinutes: '15',
  });
  const [submitting, setSubmitting] = useState(false);
  const [locating, setLocating] = useState(false);
  const [lookingUp, setLookingUp] = useState(false);
  const [areaNote, setAreaNote] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [fieldErrors, setFieldErrors] = useState<FieldErrors>({});
  // Fields the owner typed in themselves: a later lookup must never overwrite these.
  const edited = useRef(new Set<string>());

  const set = (k: keyof typeof f) => (e: React.ChangeEvent<HTMLInputElement>) => {
    edited.current.add(k);
    setF((p) => ({ ...p, [k]: e.target.value }));
  };

  const coordsValid =
    f.latitude.trim() !== '' && f.longitude.trim() !== '' &&
    Number.isFinite(Number(f.latitude)) && Number.isFinite(Number(f.longitude));

  async function lookupArea(lat: number, lng: number) {
    setLookingUp(true);
    setAreaNote(null);
    try {
      const place = await reverseGeocode(lat, lng);
      setF((p) => {
        const next = { ...p };
        for (const k of AREA_FIELDS) {
          if (!edited.current.has(k) && place[k]) next[k] = place[k] as string;
        }
        return next;
      });
      if (!place.district && !place.state) setAreaNote("We couldn't find your district automatically. Please type it in.");
    } catch {
      setAreaNote("We couldn't find your district automatically. Please type it in.");
    } finally {
      setLookingUp(false);
    }
  }

  function useMyLocation() {
    if (!navigator.geolocation) return setError('This browser cannot share your location. Enter it below.');
    setLocating(true);
    setError(null);
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        const lat = pos.coords.latitude;
        const lng = pos.coords.longitude;
        setF((p) => ({ ...p, latitude: lat.toFixed(6), longitude: lng.toFixed(6) }));
        setLocating(false);
        void lookupArea(lat, lng);
      },
      () => {
        setLocating(false);
        setError('Location unavailable. Please type the latitude and longitude of your salon.');
      },
      { enableHighAccuracy: true, timeout: 10000 },
    );
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setSubmitting(true);
    setError(null);
    setFieldErrors({});
    try {
      const district = f.district.trim();
      const state = f.state.trim();
      await registerSalon({
        salonName: f.salonName,
        phone: f.phone,
        address: f.address,
        latitude: Number(f.latitude),
        longitude: Number(f.longitude),
        ...(district && { district }),
        ...(state && { state }),
        ownerName: f.ownerName,
        email: f.email,
        password: f.password,
        openingTime: f.openingTime,
        closingTime: f.closingTime,
        averageServiceMinutes: Number(f.averageServiceMinutes),
      });

    } catch (err) {

        setError(err.message);
        setFieldErrors(err.fieldErrors ?? {});
      } else {
        setError('We could not reach the server. Please check your connection and try again.');
      }
    } finally {
      setSubmitting(false);
    }
  }

  const fieldError = (k: string) =>
    fieldErrors[k]?.[0] ? (
      <p id={`${k}-error`} className="mt-1 text-sm text-red-700">{fieldErrors[k]![0]}</p>
    ) : null;

  const input = (

      <input
        id={k} className="field" value={f[k]} onChange={set(k)} required
        aria-invalid={fieldErrors[k] ? true : undefined}
        aria-describedby={fieldErrors[k] ? `${k}-error` : undefined}
        {...extra}
      />
      {fieldError(k)}
    </div>
  );

  return (
    <PageShell title="Register your salon">
      <BrandMark />
      <h1 className="mb-1 text-2xl font-semibold">Register your salon</h1>
      <p className="mb-5 text-sm text-gray-600">
        Set up your queue in a minute. You'll get a QR code to print and stick up. Customers never need an account.
      </p>
      <form onSubmit={handleSubmit} className="space-y-4" noValidate>
        <section className="card space-y-4 p-5">

          {input('salonName', 'Salon name', { autoComplete: 'organization' })}
          {input('phone', 'Salon phone number', { type: 'tel', inputMode: 'tel', autoComplete: 'tel' })}
          {input('address', 'Address', { autoComplete: 'street-address' })}
        </section>

        <section className="card space-y-4 p-5">

            {input('latitude', 'Latitude', { inputMode: 'decimal' })}

function ProfileForm({ salon }: { salon: BarberSalonProfile }) {
  const queryClient = useQueryClient();
  const [name, setName] = useState(salon.name);
  const [phone, setPhone] = useState(salon.phone);
  const [address, setAddress] = useState(salon.address);
  const [opening, setOpening] = useState(salon.openingTime);
  const [closing, setClosing] = useState(salon.closingTime);
  const [minutes, setMinutes] = useState(String(salon.averageServiceMinutes));
  const [msg, setMsg] = useState<{ ok: boolean; text: string } | null>(null);

  // Only what actually changed is sent.
  const changes: SalonProfileUpdate = {};
  if (name.trim() !== salon.name) changes.name = name.trim();
  if (phone.trim() !== salon.phone) changes.phone = phone.trim();
  if (address.trim() !== salon.address) changes.address = address.trim();
  if (opening !== salon.openingTime) changes.openingTime = opening;
  if (closing !== salon.closingTime) changes.closingTime = closing;
  if (minutes !== '' && Number(minutes) !== salon.averageServiceMinutes) {
    changes.averageServiceMinutes = Number(minutes);
  }
  const dirty = Object.keys(changes).length > 0;

  const save = useMutation({
    mutationFn: (c: SalonProfileUpdate) => updateBarberSalonProfile(c),
    onSuccess: (updated) => {
      // The server may have tidied values (e.g. phone), so show what it stored.
      setName(updated.name);
      setPhone(updated.phone);
      setAddress(updated.address);
      setOpening(updated.openingTime);
      setClosing(updated.closingTime);
      setMinutes(String(updated.averageServiceMinutes));
      queryClient.setQueryData(PROFILE_KEY, updated);
      queryClient.invalidateQueries({ queryKey: ['barberSalon'] }); // QR poster shows the name
      setMsg({ ok: true, text: 'Saved.' });
    },
    onError: (e: unknown) => setMsg({ ok: false, text: e instanceof Error ? e.message : 'Could not save.' }),
  });

  function submit(e: React.FormEvent) {
    e.preventDefault();
    setMsg(null);
    const m = Number(minutes);
    if (!Number.isInteger(m) || m < 1 || m > 240) {
      return setMsg({ ok: false, text: 'Average service time must be 1 to 240 minutes.' });
    }
    if (!opening || !closing || opening >= closing) {
      return setMsg({ ok: false, text: 'Closing time must be after opening time.' });
    }
    if (dirty) save.mutate(changes);
  }

  const touch = () => setMsg(null);

  return (

          onChange={(e) => { setPhone(e.target.value); touch(); }} />
      </div>
      <div>
        <label htmlFor="salon-address" className="mb-1 block text-sm font-medium">Address</label>

      </div>
      <div className="grid grid-cols-2 gap-3">
        <div>
          <label htmlFor="salon-opening" className="mb-1 block text-sm font-medium">Opens at</label>
          <input id="salon-opening" type="time" className="field" value={opening}
            onChange={(e) => { setOpening(e.target.value); touch(); }} />
        </div>
        <div>
          <label htmlFor="salon-closing" className="mb-1 block text-sm font-medium">Closes at</label>

        </div>
      </div>
      <div>
        <label htmlFor="salon-minutes" className="mb-1 block text-sm font-medium">
          Average time per customer (minutes)
        </label>
        <input id="salon-minutes" className="field" inputMode="numeric" value={minutes}
          onChange={(e) => { setMinutes(e.target.value.replace(/\D/g, '').slice(0, 3)); touch(); }} />

        </p>
      )}

import { ApiError } from '../api/client';

type Period = 'today' | 'week' | 'month';

const PERIODS: { id: Period; label: string }[] = [
  { id: 'today', label: 'Today' },
  { id: 'week', label: 'This week' },
  { id: 'month', label: 'This month' },
];

// Dates arrive as IST calendar days (YYYY-MM-DD). Build them from parts so the
// phone's own timezone can never shift the day.
function parseYmd(ymd: string) {
  const [y, m, d] = ymd.split('-').map(Number);
  return new Date(y, m - 1, d);
}
const fmtWeekday = (ymd: string) => parseYmd(ymd).toLocaleDateString('en-IN', { weekday: 'short' });
const fmtShort = (ymd: string) => parseYmd(ymd).toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
const fmtFull = (ymd: string) =>
  parseYmd(ymd).toLocaleDateString('en-IN', { weekday: 'short', day: 'numeric', month: 'short' });
const customers = (n: number) => `${n} ${n === 1 ? 'customer' : 'customers'}`;

function viewFor(stats: BarberStats, period: Period) {
  if (period === 'today') {
    return {
      total: stats.servedToday,
      qr: stats.qrToday,
      walkIn: stats.walkInToday,
      prev: stats.servedYesterday,
      noun: 'today',
      vs: 'yesterday',
      days: null as StatsDay[] | null,
    };
  }
  const p = period === 'week' ? stats.week : stats.month;
  return {
    total: p.total,
    qr: p.qr,
    walkIn: p.walkIn,
    prev: p.prevSamePeriod,
    noun: period === 'week' ? 'this week' : 'this month',
    vs: period === 'week' ? 'the same days last week' : 'the same days last month',
    days: p.days,
  };
}

// Words and a symbol, never colour alone.
function deltaOf(total: number, prev: number, vs: string) {
  const d = total - prev;
  if (d === 0) return { symbol: '=', text: `Same as ${vs}`, tone: 'text-gray-700' };
  if (d > 0) return { symbol: '▲', text: `${d} more than ${vs}`, tone: 'text-green-800' };
  return { symbol: '▼', text: `${Math.abs(d)} fewer than ${vs}`, tone: 'text-gray-700' };
}

function DayBars({
  days,
  today,
  showCounts,
  label,
}: {
  days: StatsDay[];
  today: string;
  showCounts: boolean;
  label: string;
}) {
  const max = Math.max(1, ...days.map((d) => d.count));
  return (
    <>
      <ul role="list" aria-label={label} className={`flex items-end ${showCounts ? 'gap-2' : 'gap-0.5'}`}>
        {days.map((d) => {
          const future = d.date > today;
          const isToday = d.date === today;
          const pct = d.count > 0 ? Math.max(6, Math.round((d.count / max) * 100)) : 0;
          const fill = future ? 'bg-gray-200' : isToday ? 'bg-brand border-2 border-ink' : 'bg-ink';
          return (
            <li key={d.date} aria-current={isToday ? 'date' : undefined} className="flex min-w-0 flex-1 flex-col items-center">
              {showCounts && (
                <span aria-hidden="true" className="mb-1 h-4 text-xs font-bold tabular-nums text-ink">
                  {future ? '' : d.count}
                </span>
              )}
              <div aria-hidden="true" className="flex h-32 w-full items-end">
                <div className={`w-full rounded-t ${fill}`} style={{ height: `${pct}%`, minHeight: 3 }} />
              </div>
              {showCounts && (
                <span
                  aria-hidden="true"
                  className={`mt-1 text-xs ${isToday ? 'font-bold text-ink' : 'text-gray-600'}`}
                >
                  {fmtWeekday(d.date)}
                </span>
              )}
              <span className="sr-only">
                {fmtFull(d.date)}
                {isToday ? ' (today)' : ''}: {future ? 'upcoming' : customers(d.count)}
              </span>
            </li>
          );
        })}
      </ul>
      {!showCounts && days.length > 0 && (
        <div aria-hidden="true" className="mt-1 flex justify-between text-xs text-gray-600">
          <span>{fmtShort(days[0].date)}</span>
          <span>{fmtShort(days[days.length - 1].date)}</span>
        </div>
      )}
    </>
  );
}
